"""Standalone adapter, loaded ONLY by the private Office Python/UNO interpreter."""

import json
import subprocess
import sys
import time
import uuid
from pathlib import Path

import uno
import unohelper
from com.sun.star.task import XInteractionHandler


class AbortInteraction(unohelper.Base, XInteractionHandler):
    def __init__(self):
        self.requests = []

    def handle(self, request):
        continuations = request.getContinuations()
        if any(
            item.queryInterface(uno.getTypeByName("com.sun.star.task.XInteractionPassword"))
            for item in continuations
        ):
            reason = "password required (encrypted input)"
        else:
            try:
                details = request.getRequest()
                reason = getattr(details, "typeName", type(details).__name__)
            except Exception:
                # This pinned UNO bridge cannot decode some request exceptions.
                # Their abort continuation remains usable; never approve them.
                reason = "unreadable interaction request"
        self.requests.append(reason)
        for continuation in continuations:
            if hasattr(continuation, "abort"):
                continuation.abort()
                return
            interfaces = uno.getTypeByName("com.sun.star.task.XInteractionAbort")
            if continuation.queryInterface(interfaces):
                continuation.select()
                return


def properties(values):
    result = []
    for name, (kind, value) in values.items():
        item = uno.createUnoStruct("com.sun.star.beans.PropertyValue")
        item.Name = name
        item.Value = uno.Any(kind, value)
        result.append(item)
    return tuple(result)


def text_fonts(text, result):
    paragraphs = text.createEnumeration()
    while paragraphs.hasMoreElements():
        paragraph = paragraphs.nextElement()
        if paragraph.supportsService("com.sun.star.text.TextTable"):
            for name in paragraph.getCellNames():
                text_fonts(paragraph.getCellByName(name), result)
            continue
        if not hasattr(paragraph, "createEnumeration"):
            continue
        portions = paragraph.createEnumeration()
        while portions.hasMoreElements():
            portion = portions.nextElement()
            value = getattr(portion, "String", "")
            if value.strip() and not getattr(portion, "CharHidden", False):
                result.append(
                    {
                        "text": value,
                        "requested": getattr(portion, "CharFontName", ""),
                        "requested_asian": getattr(portion, "CharFontNameAsian", ""),
                        "requested_complex": getattr(portion, "CharFontNameComplex", ""),
                    }
                )


def writer_fonts(document):
    result = []

    text_fonts(document.Text, result)
    frames = document.getTextFrames()
    for index in range(frames.getCount()):
        text_fonts(frames.getByIndex(index).Text, result)
    styles = document.StyleFamilies.getByName("PageStyles")
    for name in styles.getElementNames():
        style = styles.getByName(name)
        for section in ("Header", "Footer"):
            if getattr(style, section + "IsOn", False):
                text_fonts(getattr(style, section + "Text"), result)
                if not getattr(style, section + "IsShared", True):
                    text_fonts(getattr(style, section + "TextLeft"), result)
    return result


def export_pdf(document, path, export_filter, options):
    export = properties({"FilterName": ("string", export_filter)})
    data = uno.createUnoStruct("com.sun.star.beans.PropertyValue")
    data.Name = "FilterData"
    data.Value = uno.Any("[]com.sun.star.beans.PropertyValue", properties(options))
    document.storeToURL(path.as_uri(), (*export, data))


def presentation(request, document, directory):
    detected = {value.Name: value.Value for value in document.getArgs()}
    if detected.get("FilterName") != "Impress MS PowerPoint 2007 XML":
        raise ValueError("PPTX did not match the required PowerPoint input filter")
    pages = document.getDrawPages()
    if not pages.getCount():
        raise ValueError("Presentation contains no slides")
    slides, fonts = [], []
    for index in range(pages.getCount()):
        page = pages.getByIndex(index)
        notes_page = page.getNotesPage()
        notes = []
        for number in range(notes_page.getCount()):
            shape = notes_page.getByIndex(number)
            if shape.getShapeType() == "com.sun.star.presentation.NotesShape":
                notes.append(shape.String)
        for number in range(page.getCount()):
            shape = page.getByIndex(number)
            if hasattr(shape, "createEnumeration"):
                text_fonts(shape, fonts)
        slides.append(
            {
                "ordinal": index + 1,
                "page": index + 1,
                "name": page.Name,
                "hidden": not page.Visible,
                "notes": "\n".join(notes),
            }
        )
        # Export an explicit original slide ordinal. The application verifies
        # each output has exactly one page before assembling the snapshot.
        export_pdf(
            document,
            directory / f"slide-{index + 1}.pdf",
            "impress_pdf_Export",
            {**request["pdf_options"], "PageRange": ("string", str(index + 1))},
        )
    return {
        "filter": "impress_pdf_Export",
        "detected_filter": detected["FilterName"],
        "requested_fonts": fonts,
        "slides": slides,
        "diagnostics": [
            "All slides, including hidden slides; speaker notes retain their original slide."
        ],
    }


def convert(request, directory):
    source = Path(request["input"])
    with source.open("rb") as stream:
        header = stream.read(8)
    if header == b"\xd0\xcf\x11\xe0\xa1\xb1\x1a\xe1" and source.stat().st_size < 512:
        raise ValueError("Damaged Office compound-file header is incomplete")
    pipe = "openkb_" + uuid.uuid4().hex
    process = subprocess.Popen(
        [
            request["soffice"],
            "--headless",
            "--nologo",
            "--nodefault",
            "--norestore",
            "-env:UserInstallation=" + (directory / "profile").as_uri(),
            f"--accept=pipe,name={pipe};urp;StarOffice.ComponentContext",
        ]
    )
    document = None
    desktop = None
    try:
        context = uno.getComponentContext()
        resolver = context.ServiceManager.createInstanceWithContext(
            "com.sun.star.bridge.UnoUrlResolver", context
        )
        while True:
            try:
                remote = resolver.resolve(f"uno:pipe,name={pipe};urp;StarOffice.ComponentContext")
                break
            except Exception:
                if process.poll() is not None:
                    raise RuntimeError(
                        "Office exited before its private UNO pipe was ready"
                    ) from None
                time.sleep(0.05)  # Supervisor owns the finite deadline.
        desktop = remote.ServiceManager.createInstanceWithContext(
            "com.sun.star.frame.Desktop", remote
        )
        options = properties(request["load_options"])
        interaction = uno.createUnoStruct("com.sun.star.beans.PropertyValue")
        handler = AbortInteraction()
        interaction.Name, interaction.Value = "InteractionHandler", handler
        document = desktop.loadComponentFromURL(
            Path(request["input"]).as_uri(), "_blank", 0, (*options, interaction)
        )
        if handler.requests:
            raise ValueError(
                "Office input requires unsupported interaction: " + ", ".join(handler.requests)
            )
        if document is None:
            raise ValueError("Office could not load the damaged or unsupported input")
        if Path(request["input"]).suffix.lower() == ".pptx":
            if not document.supportsService("com.sun.star.presentation.PresentationDocument"):
                raise ValueError("PowerPoint input did not load as a presentation")
            return presentation(request, document, directory)
        if not document.supportsService("com.sun.star.text.TextDocument"):
            raise ValueError("Office input did not load as a Writer document")
        settings = document.createInstance("com.sun.star.text.DocumentSettings")
        if not settings.getPropertySetInfo().hasPropertyByName("PrintHiddenText"):
            raise ValueError("Pinned Writer lacks the required hidden-text setting")
        settings.setPropertyValue("PrintHiddenText", False)
        controller = document.getCurrentController()
        controller.setPropertyValue("PDFExport_ShowChanges", False)
        if controller.getPropertyValue("PDFExport_ShowChanges"):
            raise ValueError("Writer did not apply the required final-visible revision display")
        requested_fonts = writer_fonts(document)
        detected = {value.Name: value.Value for value in document.getArgs()}
        extension = Path(request["input"]).suffix.lower()
        filters = {
            ".docx": {"Office Open XML Text", "MS Word 2007 XML"},
            ".doc": {"MS Word 97", "MS WinWord 6.0", "MS WinWord 5"},
        }
        if detected.get("FilterName") not in filters.get(extension, set()):
            raise ValueError(
                f"Office detected filter {detected.get('FilterName')!r} does not match {extension}"
            )
        export_pdf(document, directory / "output.pdf", "writer_pdf_Export", request["pdf_options"])
        return {
            "filter": "writer_pdf_Export",
            "detected_filter": str(detected.get("FilterName", "")),
            "requested_fonts": requested_fonts,
            "print_hidden_text": False,
            "diagnostics": ["Writer final visible text; comments and tracked deletions excluded."],
        }
    finally:
        if document is not None:
            document.close(True)
        if desktop is not None:
            desktop.terminate()
        # Supervisor reaps even a vetoed/blocked termination and the shell launcher.


def main():
    path = Path(sys.argv[1])
    try:
        result = convert(json.loads(path.read_text(encoding="utf-8")), path.parent)
        (path.parent / "result.json").write_text(json.dumps(result), encoding="utf-8")
        return 0
    except Exception as exc:
        (path.parent / "failure.json").write_text(
            json.dumps({"error": f"{type(exc).__name__}: {exc}"}), encoding="utf-8"
        )
        return 1


if __name__ == "__main__":
    sys.exit(main())
